cmake_minimum_required(VERSION 3.16)
project(STServo VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(stservo STATIC src/stservo.cpp)
target_include_directories(stservo PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(stservo PUBLIC cxx_std_11)
set_target_properties(stservo PROPERTIES CXX_EXTENSIONS OFF)
function(stservo_warnings target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
stservo_warnings(stservo)
add_library(stservo_examples OBJECT examples/multi_servo.cpp examples/control_setup.cpp)
target_link_libraries(stservo_examples PRIVATE stservo)
stservo_warnings(stservo_examples)
if(BUILD_TESTING)
    add_executable(stservo_tests tests/test_stservo.cpp)
    target_link_libraries(stservo_tests PRIVATE stservo)
    stservo_warnings(stservo_tests)
    add_test(NAME protocol COMMAND stservo_tests)
endif()
if(UNIX)
    find_package(Threads REQUIRED)
    add_library(stservo_posix STATIC adapters/posix/stservo_posix.cpp)
    target_include_directories(stservo_posix PUBLIC adapters/posix)
    target_link_libraries(stservo_posix PUBLIC stservo Threads::Threads)
    stservo_warnings(stservo_posix)
    add_executable(stservo_ping examples/raspberry_pi/main.cpp)
    target_link_libraries(stservo_ping PRIVATE stservo_posix)
    stservo_warnings(stservo_ping)
    add_executable(stservo_minimal examples/raspberry_pi/minimal_control.cpp)
    target_link_libraries(stservo_minimal PRIVATE stservo_posix)
    stservo_warnings(stservo_minimal)
    if(BUILD_TESTING)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME linux_pty COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_posix.py $<TARGET_FILE:stservo_ping>)
        add_test(NAME minimal_control_pty COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_minimal_control.py $<TARGET_FILE:stservo_minimal>)
    endif()
endif()
find_package(Doxygen QUIET)
if(DOXYGEN_FOUND)
    add_custom_target(docs COMMAND ${DOXYGEN_EXECUTABLE} Doxyfile
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        COMMENT "Generate STServo API documentation")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../stservo/include/stm32_stservo_transport.hpp")
        add_custom_target(docs-stm32 COMMAND ${DOXYGEN_EXECUTABLE} Doxyfile.stm32
            WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
            COMMENT "Generate API documentation including the workspace STM32 adapter")
    endif()
endif()
